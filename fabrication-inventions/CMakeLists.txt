cmake_minimum_required(VERSION 3.20)
project(twin_fabrication_inventions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(twin_fabrication src/main.cpp)
if(MSVC)
  target_compile_options(twin_fabrication PRIVATE /O2 /W4)
else()
  target_compile_options(twin_fabrication PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
